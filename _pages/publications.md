<script>
(function () {
  function updatePublications() {
    const pub = document.querySelector(".publications");
    if (!pub) return;

    /*
     * Highlight Miyeon Jung without rewriting the bibliography HTML.
     * Existing links, buttons, dates, and event handlers are preserved.
     */
    pub.querySelectorAll(".author, .authors").forEach(function (block) {
      const walker = document.createTreeWalker(
        block,
        NodeFilter.SHOW_TEXT,
        {
          acceptNode: function (node) {
            const parent = node.parentElement;

            if (
              !parent ||
              parent.closest(".my-name, script, style")
            ) {
              return NodeFilter.FILTER_REJECT;
            }

            return NodeFilter.FILTER_ACCEPT;
          }
        }
      );

      const nodes = [];
      while (walker.nextNode()) {
        nodes.push(walker.currentNode);
      }

      nodes.forEach(function (node) {
        const text = node.nodeValue;
        const pattern =
          /\b(?:Miyeon|M\.)\s+Jung\b[†*]?|\bJung[†*]?,\s*(?:Miyeon\b|M\.)/g;

        const matches = Array.from(text.matchAll(pattern));
        if (!matches.length) return;

        const fragment = document.createDocumentFragment();
        let position = 0;

        matches.forEach(function (match) {
          fragment.appendChild(
            document.createTextNode(
              text.slice(position, match.index)
            )
          );

          const name = document.createElement("span");
          name.className = "my-name";
          name.textContent = match[0];
          fragment.appendChild(name);

          position = match.index + match[0].length;
        });

        fragment.appendChild(
          document.createTextNode(text.slice(position))
        );

        node.parentNode.replaceChild(fragment, node);
      });
    });

    /*
     * Ensure manuscript status remains visible even when the
     * bibliography template does not display the BibTeX note field.
     */
    const statuses = [
      {
        key: "jung2026untapped",
        title: "The Untapped Power of Wishlists",
        text: "Conditionally accepted"
      },
      {
        key: "jung2026humanai",
        title: "Interaction and Traffic Safety in Semi-Autonomous Driving",
        text: "Major revision"
      },
      {
        key: "jung2026digital",
        title: "Digital Access Restrictions and Creator Responses",
        text: "Major revision"
      },
      {
        key: "jung2026social",
        title: "Social Media Activism and the Legitimization",
        text: "Under review"
      },
      {
        key: "bhargava2026goldilocks",
        title: "A Goldilocks Effect in Recommendation Novelty",
        text: "To be presented at WISE 2026"
      },
      {
        key: "jung2026starting",
        title: "Starting Farther from the Goal",
        text: "Field study of membership goal pursuit"
      },
      {
        key: "baek2026fair",
        title: "Fair Algorithms, Fair Decisions?",
        text: "Pre-registered experiment"
      }
    ];

    const items = Array.from(
      pub.querySelectorAll("ol.bibliography > li")
    );

    statuses.forEach(function (status) {
      const reference = document.getElementById(status.key);
      let item = reference ? reference.closest("li") : null;

      if (!item || !pub.contains(item)) {
        item = items.find(function (candidate) {
          const title = candidate.querySelector(".title");

          return title &&
            title.textContent.includes(status.title);
        });
      }

      if (!item) return;

      const visibleText = item.innerText || "";

      if (
        visibleText.toLowerCase().includes(
          status.text.toLowerCase()
        )
      ) {
        return;
      }

      if (item.querySelector(".publication-status")) return;

      const label = document.createElement("div");
      label.className = "publication-status";
      label.textContent = status.text;

      const periodical = item.querySelector(".periodical");
      const title = item.querySelector(".title");

      if (periodical) {
        periodical.insertAdjacentElement("afterend", label);
      } else if (title) {
        title.parentElement.appendChild(label);
      } else {
        const content = item.querySelector(".row") ||
          item.firstElementChild;

        if (content) content.appendChild(label);
      }
    });
  }

  if (document.readyState === "loading") {
    document.addEventListener(
      "DOMContentLoaded",
      updatePublications
    );
  } else {
    updatePublications();
  }
})();
</script>
